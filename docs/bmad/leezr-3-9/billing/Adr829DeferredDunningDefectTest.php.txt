<?php

namespace Tests\Feature;

use App\Core\Billing\DunningEngine;
use App\Core\Billing\InvoiceIssuer;
use App\Core\Billing\Subscription;
use App\Core\Models\Company;
use App\Core\Plans\PlanRegistry;
use App\Jobs\Billing\ProcessDunningBatchJob;
use Carbon\Carbon;
use Illuminate\Foundation\Testing\RefreshDatabase;
use Tests\TestCase;

/**
 * ADR-829 (D-17) — PREPARED, NOT IN THE SUITE. Fails on the current code, passes with
 * deferred-dunning-guard.patch. A ProcessDunningBatchJob consumed late (after the daily scan already
 * retried the invoice) or delivered twice must not make an extra attempt : an extra attempt shortens the
 * policy (1, 3, 7 days) and can mark the invoice uncollectible — and suspend the company — days early.
 */
class Adr829DeferredDunningDefectTest extends TestCase
{
    use RefreshDatabase;

    public function test_a_late_or_duplicated_job_does_not_shorten_the_retry_policy(): void
    {
        $this->seed(\Database\Seeders\PlatformSeeder::class);
        PlanRegistry::sync();
        $company = Company::create(['name' => 'Late', 'slug' => 'late829', 'plan_key' => 'pro', 'status' => 'active', 'jobdomain_key' => 'logistique']);
        $sub = Subscription::create(['company_id' => $company->id, 'plan_key' => 'pro', 'interval' => 'monthly', 'status' => 'active',
            'current_period_start' => Carbon::parse('2026-03-01'), 'current_period_end' => Carbon::parse('2026-03-31')]);
        Carbon::setTestNow(Carbon::parse('2026-03-10'));
        $invoice = InvoiceIssuer::finalize(tap(InvoiceIssuer::createDraft($company, $sub->id), fn ($i) => InvoiceIssuer::addLine($i, 'plan', 'Pro', 2900)));
        $invoice->update(['due_at' => Carbon::parse('2026-03-05')]);
        DunningEngine::processOverdueInvoices();

        // 03-11 : the async run dispatched a job (no consumer) ; the synchronous scan then retried the invoice.
        Carbon::setTestNow(Carbon::parse('2026-03-11 01:00:00'));
        $job = new ProcessDunningBatchJob(collect([$invoice->id]));
        DunningEngine::processOverdueInvoices();
        $this->assertSame(1, $invoice->fresh()->retry_count);

        // 03-12 : the job is consumed late, then delivered a second time.
        Carbon::setTestNow(Carbon::parse('2026-03-12 09:00:00'));
        $job->handle();
        $job->handle();

        $invoice->refresh();
        $this->assertSame(1, $invoice->retry_count, 'no attempt before next_retry_at (2026-03-14)');
        $this->assertSame('overdue', $invoice->status);
        $this->assertSame('active', $company->fresh()->status, 'no premature suspension');
        Carbon::setTestNow();
    }
}
